% =============================================================================
% LaTeX preamble for nox-mem paper (arXiv-safe, xelatex)
%
% Monospace: default Latin Modern Mono (from lmodern) -- bundled with every
% TeX Live, including arXiv's AutoTeX. We deliberately do NOT \setmonofont a
% system font: Courier New / DejaVu Sans Mono are absent on arXiv's Linux
% TeX Live and would break the xelatex build (fontspec "font not found").
%
% Trade-off: a handful of rare Unicode math glyphs (e.g. inside code blocks)
% may fall back to a box; these are compiler warnings, not errors, and prose
% math renders correctly via unicode-math.
% =============================================================================

\usepackage{fontspec}
\usepackage{amssymb}
\usepackage{amsmath}

% ---------------------------------------------------------------------------
% Compact wide tables: shrink all longtables + tighten column padding so
% multi-column comparison tables (e.g. the 9-column "Six Gaps") don't squeeze
% a prose column down to one word per line. All arXiv-safe (etoolbox in TeX Live).
% ---------------------------------------------------------------------------
\usepackage{etoolbox}
\AtBeginEnvironment{longtable}{\footnotesize}
\setlength{\tabcolsep}{4pt}

% ---------------------------------------------------------------------------
% A CAUSA de todo glifo perdido: `fontspec` carregado e nunca instruido.
%
% O template do pandoc faz \usepackage{fontspec} sob XeTeX mas NAO chama
% \setmainfont, e sem isso o fontspec mantem a Latin Modern *Type1/EC*, de 256
% slots. Medido em 2026-09-11 neste manuscrito: 63 "Missing character" em
% ec-lmr10, ec-lmr8, ec-lmbx8, ec-lmtt8, ec-lmtt10 — os glifos alpha, seta,
% en dash, em dash, reticencias, menos, conjunto vazio, intersecao, aproximado
% e maior-ou-igual. Cada um **desaparece do PDF sem erro**: o build sai 0 e o
% caractere nao esta la.
%
% A Latin Modern OpenType traz todos e vem na TeX Live. Carregada pelo NOME DO
% ARQUIVO porque por nome de familia o fontspec nao a encontra nesta
% instalacao ("The font ... cannot be found").
%
% Isto substitui quatro remendos que eu tinha empilhado um a um, cada um a
% revelar o glifo seguinte: era UM defeito, nao cinco.
% ---------------------------------------------------------------------------

  BoldFont       = lmroman10-bold.otf,
  ItalicFont     = lmroman10-italic.otf,

% ---------------------------------------------------------------------------
% Pagination, measured on v1.0: of 25 headings, one ("6. What would falsify each
% claim") landed on the LAST line of page 3, with its section starting overleaf.
% A heading is a promise that text follows it; at the foot of a page it promises
% nothing. needspace is already loaded for tables, so the same mechanism serves
% here: a heading demands room for itself plus a few lines of its own section,
% or it moves to the next page with them.
%
% 13 lines, not 4, and the number was MEASURED, not chosen: four of the 25
% headings are followed immediately by a table, and a table asks for 10 lines of
% its own (above). At 4 the heading's demand was satisfied, the table's was not,
% and the table left WITHOUT the heading — which is how "9. Cost" and
% "11.3 Controls" ended up alone at the foot. Calibrated against the opposite
% defect, pages left half empty by a demand that is too large: 4 -> 2 stranded /
% 5 short pages, 8 -> 1 / 6, 13 -> 0 / 5, 16 -> 0 / 7. 13 is the only value that
% clears the stranding without costing a page.
%
% \pretocmd rather than titlesec: titlesec would also take over how the headings
% LOOK, and the look is the one thing about this layout that already matches the
% deposited papers. etoolbox is loaded above.
% ---------------------------------------------------------------------------
\pretocmd{\section}{\needspace{13\baselineskip}}{}{\PackageWarning{prereg}{section hook failed}}
\pretocmd{\subsection}{\needspace{12\baselineskip}}{}{\PackageWarning{prereg}{subsection hook failed}}
\pretocmd{\subsubsection}{\needspace{11\baselineskip}}{}{\PackageWarning{prereg}{subsubsection hook failed}}

% A single line left alone at the top (widow) or foot (club) of a page. 10000 is
% the "never" end of the scale; the emergencystretch above absorbs the slack this
% creates, which is why it does not produce short pages here.
\widowpenalty=10000
\clubpenalty=10000
\displaywidowpenalty=10000
% Do not hyphenate the last word of a page: the reader has to turn the page to
% finish a word.
\brokenpenalty=10000
